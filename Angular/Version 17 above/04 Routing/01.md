```typescript
import { Routes } from '@angular/router';

export const routes: Routes = [
    {
        path: '', pathMatch: 'full', loadComponent: ()=> { return import("./home/home.component").then( x => x.HomeComponent)}
    },
    { path: 'todos', loadComponent: ()=> { return import("./todos/todos.component").then( x => x.TodosComponent)} },
];
```

3. browser  
```url
http://localhost:4200/
<!-- will render home component by default-->
```
```url
http://localhost:4200/todos
<!-- will render todos component -->
```  
using browser url string it will reload page, which defeats purpose of SPA. we will add the `routerLinks` to Header of the page  

4. create/use element to travers the component using routerlink
`src\app\components\header\header.component.ts`:  
```typescript
import { Component, signal } from '@angular/core';
import { RouterLink } from '@angular/router';

@Component({
  selector: 'app-header',
  imports: [RouterLink], // 🔄: import & use routerlink
  templateUrl: './header.component.html',
  styleUrl: './header.component.scss'
})
export class HeaderComponent {
  title = signal("My angular header");
}
```  
`src\app\components\header\header.component.html`:  
```html
<Header>
    <nav> 
        <span routerLink="/">{{ title() }}</span> <!-- will route to home -->
        <ul>
            <li routerLink="/todos">Todos</li>  <!-- will route to todos -->
        </ul>
    </nav>
</Header>
```
